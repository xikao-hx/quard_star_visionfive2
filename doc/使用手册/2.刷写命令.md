# 刷写命令

```bash
sudo cp work/target/amp/u-boot-amp-spl.bin.normal.out \
        work/target/amp/visionfive2_fw_payload_amp.img \
        work/target/amp/image.fit \
        work/target/amp/rootfs.ext4 \
        /srv/tftp/
```

## Bootloader 启动

### AMP 启动

```bash
tftpboot 0xa0000000 u-boot-amp-spl.bin.normal.out 
mtd erase spl_a 0x0 0x40000
mtd write spl_a 0xa0000000 0x0 $filesize
mtd erase spl_b 0x0 0x40000
mtd write spl_b 0xa0000000 0x0 $filesize

tftpboot 0xa0000000 visionfive2_fw_payload_amp.img
mtd erase fw_payload_a 0x0 0x600000
mtd write fw_payload_a 0xa0000000 0x0 $filesize
mtd erase fw_payload_b 0x0 0x600000
mtd write fw_payload_b 0xa0000000 0x0 $filesize

tftpboot 0xa0000000 gpt.img
mtd erase gpt 0x0 0x40000
mtd write gpt 0xa0000000 0x0 $filesize

tftpboot 0xa0000000 recovery.img
mtd erase recovery 0x0 0x40000
mtd write recovery 0xa0000000 0x0 $filesize
```

### 一般启动

```bash
sf probe
tftpboot ${loadaddr} u-boot-amp-spl.bin.normal.out
sf update ${loadaddr} 0x000000 ${filesize}
sf update ${loadaddr} 0x040000 ${filesize}

tftpboot ${loadaddr} gpt.img
sf update ${loadaddr} 0x080000 ${filesize}

tftpboot ${loadaddr} recovery.img
sf update ${loadaddr} 0x0c0000 ${filesize}

tftpboot ${loadaddr} visionfive2_fw_payload_amp.img
sf update ${loadaddr} 0x100000 ${filesize}
sf update ${loadaddr} 0x700000 ${filesize}
```

## Linux 启动
### AMP 启动

```bash
# sdb3 和 sdb5
sudo mount /dev/sdb3 /mnt
sudo cp work/target/amp/image.fit /mnt/image.fit
sudo cp conf/jh7110_uEnv.txt conf/vf2_uEnv.txt  /mnt   
sync
sudo umount /mnt

# sdb4 和 sdb6
sudo dd if=work/target/amp/rootfs.ext4 \
    of=/dev/sdb4 bs=4M conv=fsync status=progress

# 1.建议使用自动启动，修改 initramfs 的 init 支持识别 uboot 设置的 bootargs 中的 root=PARTUUID
# 而且要支持 A/B 分区启动必须要这个
setenv bootcmd 'run sdk_boot_env; run distro_boot_env'
setenv bootargs
saveenv

# 2.uboot 直接设置 bootargs 和 bootcmd
setenv bootargs 'console=ttyS0,115200 earlycon=sbi root=/dev/mmcblk1p4 rootwait rw'
setenv bootcmd 'fatload mmc 1:3 ${loadaddr} /image.fit; bootm ${loadaddr}'   # 注意不同的格式命令不同：fatload、ext4load
saveenv
reset
```

## SD 卡分区制作

```bash
# sdb5: 100MB  sdb6: 2GB
sudo umount /dev/sdb?* 2>/dev/null || true
sudo sgdisk -e /dev/sdb

sudo sgdisk \
    -n 5:0:+100M  -t 5:0700 -c 5:boot_b \
    -n 6:0:+2048M -t 6:8300 -c 6:rootfs_b \
    /dev/sdb

sudo partprobe /dev/sdb
sudo udevadm settle

# 格式化 sdb5
sudo umount /dev/sdb5 2>/dev/null || true
sudo mkfs.vfat -F 32 -n BOOT_B /dev/sdb5
sudo blkid /dev/sdb5
```
