# OTA升级

**前序准备：**

```bash
# 生成包
./build.sh sdk build amp

./build.sh package \
  --rollback-index 10204 \
  --session-id ota-step2-20261004-003 \
  --target-bank b \
  --include-stage2

# 拷贝到 userdata 分区
rm -rf /userdata/ota/ota-step2-20261004-001
cp -a /mnt/ota_package \
  /userdata/ota/ota-step2-20261004-001
sync

# 安装驱动，创建符号链接
cd /mnt/ota
./ota-test.sh
./create_link.sh
```

**ota_info：**

```bash
ota_info dump-state
ota_info read current_bank
ota_info read target_bank
ota_info read usable_bank
ota_info read ota_state
ota_info write target_bank <a|b>
ota_info write usable_bank <a|b|ab>
ota_info reset
ota_info sync
ota_info repair
ota_info flash <file> <partition>
```

**ota_package：**

```bash
PKG=/userdata/ota/ota-step2-20261004-001
PUBKEY=/etc/ota/keys/ota_public.pem
ota_package verify_stage1     "$PKG"     "$PUBKEY"
ota_package apply_stage1      "$PKG"     "$PUBKEY"
ota_package commit_stage1     "$PKG"     "$PUBKEY"
reboot -f
ota_package verify_stage2     "$PKG"     "$PUBKEY"
ota_package apply_stage2      "$PKG"     "$PUBKEY"
ota_package commit_stage2     "$PKG"     "$PUBKEY"
reboot -f
ota_package boot_success "$PKG" "$PUBKEY"   # complete

ota_info dump-state
```

**ota_client：**

```bash
bash ota_server/scripts/publish_static_release.sh \
  --source-dir nfs_rootfs/ota_package \
  --release-id 20261005-quard-star-r5 \
  --board-type quard-star \
  --public-base-url http://192.168.5.11:18080
sudo bash ota_server/scripts/deploy_nginx_http.sh
sudo systemctl reload nginx

# 启动 FastAPI 网页后台
./ota_server/scripts/run_server.sh
```

```bash
SERVER_IP=192.168.31.2
MAC_ID=$(tr -d ':' </sys/class/net/eth0/address)

mkdir -p /userdata/ota
printf 'device_id=vf2-%s\nboard_type=quard-star\nhardware_version=visionfive2\n' "$MAC_ID" >/userdata/ota/device_identity.test.conf
cp /etc/ota/client.conf /userdata/ota/client.test.conf
sed -i \
  -e "s|^server_url=.*|server_url=http://$SERVER_IP:18080|" \
  -e "s|^manifest_url=.*|manifest_url=http://$SERVER_IP:18080/ota/manifests/quard-star.json|" \
  -e 's|^api_token=.*|api_token=ota-demo-token|' \
  /userdata/ota/client.test.conf

export OTA_CLIENT_IDENTITY_CONFIG=/userdata/ota/device_identity.test.conf
export OTA_CLIENT_CONFIG=/userdata/ota/client.test.conf

ota_client register &&
ota_client check &&
ota_client download &&
ota_client unpack &&
ota_client install

export OTA_CLIENT_IDENTITY_CONFIG=/userdata/ota/device_identity.test.conf
export OTA_CLIENT_CONFIG=/userdata/ota/client.test.conf
ota_client service_start
```
